% Borrar variables, cerrar figuras y limpiar la consola
clear all; close all; clc;

fprintf('Simulación de Cryoprobe ´para Bioimpedancia con Enfoque en PI\n\n');

% ------------------------------------------
% PARÁMETROS DE DISEÑO
% ------------------------------------------

% Frecuencias de análisis
f_min = 1e3;                % 1 kHz
f_max = 1e6;                % 1 MHz
f_work = 50e3;              % 50 kHz (frecuencia de trabajo)

% Dimensiones PCB (de FreeCAD/KiCad)
L_pcb = 60.96e-3;           % m - longitud PCB
W_pcb = 25.4e-3;            % m - ancho PCB
h_pcb = 1.69e-3;            % m - grosor PCB FR4

% Dimensiones Microstrip
L_trace = 45e-3;            % m - longitud traza
W_trace = 2.5e-3;           % m - ancho traza

% Materiales
eps_r = 4.3;                % FR4 permitividad relativa
sigma_cu = 58e6;            % S/m - Conductividad cobre
sigma_fr4 = 0.02;           % S/m - Conductividad FR4
t_copper = 35e-6;           % m - grosor cobre
tan_delta = 0.025;          % Factor de pérdidas FR4

% Constantes físicas
c0 = 3e8;           % m/s
mu0 = 4*pi*1e-7;    % H/m
eps0 = 8.85e-12;    % F/m

fprintf('Configuración del análisis:\n');
fprintf('- PCB: %.2f × %.2f × %.2f mm\n', L_pcb*1000, W_pcb*1000, h_pcb*1000);
fprintf('- Microstrip: %.1f × %.2f mm\n', L_trace*1000, W_trace*1000);
fprintf('- Metodología: Analítico + FDTD inteligente\n');
fprintf('- Timeout FDTD: 3 minutos (auto-continúa)\n\n');

%% ========================================================================
%% PARTE I: ANÁLISIS ANALÍTICO (SIEMPRE FUNCIONA)
%% ========================================================================

fprintf('PARTE I: ANÁLISIS ANALÍTICO (IPC-2141A)\n');
fprintf('=========================================\n');

% Frecuencias en distribución logarítmica
freq_analytical = logspace(log10(f_min), log10(f_max), 200);

% Relación geométrica entre el ancho del trazo y la altura del dieléctrico
w_h = W_trace / h_pcb;

% Permitividad efectiva (medio heterogéneo de dieléctrico y aire)
eps_eff = (eps_r + 1)/2 + (eps_r - 1)/2 * 1/sqrt(1 + 12/w_h);

% Impedancia característica
Z0_analytical = (120*pi) / (sqrt(eps_eff) * (w_h + 1.393 + 0.667*log(w_h + 1.444)));

% Pérdidas en el dieléctrico
alpha_d = (pi * freq_analytical' * sqrt(eps_r) * tan_delta) ./ (c0 * sqrt(eps_eff));

% Pérdidas en el conductor
omega = 2*pi*freq_analytical';                    % Frecuencia angular (rad/s)
delta_skin = sqrt(2./(omega * mu0 * sigma_cu));   % Profundidad de penetración skin (m)
alpha_c = 1./(2 * delta_skin * W_trace);          % Atenuación conductiva (Nepers/m)
alpha_total = alpha_d + alpha_c;

beta = 2*pi*freq_analytical' * sqrt(eps_eff) / c0;
gamma = alpha_total + 1j*beta;

% Coeficiente de reflexión de la carga (50Ω)
ZL = 50; % Carga
Gamma_L = (ZL - Z0_analytical) / (ZL + Z0_analytical);

% Parámetros S de línea de transmisión
S11_analytical = Gamma_L * (1 - exp(-2*gamma*L_trace)) ./ (1 - Gamma_L^2 * exp(-2*gamma*L_trace));
S21_analytical = (1 - Gamma_L^2) * exp(-gamma*L_trace) ./ (1 - Gamma_L^2 * exp(-2*gamma*L_trace));

% Impedancia de entrada
Z_in_analytical = Z0_analytical * (ZL + Z0_analytical*tanh(gamma*L_trace)) ./ (Z0_analytical + ZL*tanh(gamma*L_trace));

% VSWR
VSWR_analytical = (1 + abs(S11_analytical)) ./ (1 - abs(S11_analytical));

fprintf('Permitividad efectiva: %.2f\n', eps_eff);
fprintf('Impedancia característica: %.1f Ω\n', Z0_analytical);

%% 3. SIMULACIÓN FDTD (OpenEMS)
fprintf('\n=== SIMULACIÓN FDTD ===\n');

% Verificar disponibilidad de OpenEMS
CSX = InitCSX();

% 3.1 Configuración FDTD
lambda_min = c0 / (f_max * sqrt(eps_r)); % Longitud de onda mínima
resolution = lambda_min / 8;  % Tamaño de celda

% Inicializar FDTD
% Establecer timesteps y criterio de convergencia
FDTD = InitFDTD('NrTS', 8000, 'EndCriteria', 1e-3);
% Crear un pulso gaussiano
FDTD = SetGaussExcite(FDTD, 0.2*(f_min+f_max), 0.3*(f_max-f_min));
% Condiciones de Frontera
BC = {'PML_8' 'PML_8' 'PML_8' 'PML_8' 'PML_8' 'PML_8'};
FDTD = SetBoundaryCond(FDTD, BC);

% Malla
pml_size = 8*resolution;
mesh.x = [-W_pcb/2-pml_size : resolution*1.5 : W_pcb/2+pml_size];
mesh.y = [0-pml_size : resolution*1.5 : L_pcb+pml_size];
mesh.z = [-h_pcb-pml_size : resolution : 2*resolution+pml_size];
CSX = DefineRectGrid(CSX, 1e-3, mesh);

% Materiales
CSX = AddMaterial(CSX, 'FR4');
CSX = SetMaterialProperty(CSX, 'FR4', 'Epsilon', eps_r);
CSX = AddMetal(CSX, 'copper');

% Geometría de la PCB (Coordenadas de Kicad)
% Substrato
start = [-W_pcb/2, 0, -h_pcb];
stop  = [W_pcb/2, L_pcb, 0];
CSX = AddBox(CSX, 'FR4', 0, start, stop);

% Plano de tierra
start = [-W_pcb/2, 0, -h_pcb-t_copper];
stop  = [W_pcb/2, L_pcb, -h_pcb];
CSX = AddBox(CSX, 'copper', 0, start, stop);

% Microstrip
start = [-W_trace/2, (L_pcb-L_trace)/2, 0];
stop  = [W_trace/2, (L_pcb+L_trace)/2, t_copper];
CSX = AddBox(CSX, 'copper', 0, start, stop);

% Puertos
% Puerto de Entrada
start = [-W_trace/2, (L_pcb-L_trace)/2, 0];
stop  = [W_trace/2, (L_pcb-L_trace)/2, h_pcb];
[CSX, port{1}] = AddLumpedPort(CSX, 5, 1, 50, start, stop, [0 0 1], true);

% Puerto de Salida
start = [-W_trace/2, (L_pcb+L_trace)/2, 0];
stop  = [W_trace/2, (L_pcb+L_trace)/2, h_pcb];
[CSX, port{2}] = AddLumpedPort(CSX, 5, 2, 50, start, stop, [0 0 1]);

% Ejecutar simulación
Sim_Path = 'tmp_microstrip'; % Directorio temporal de simulación
Sim_CSX = 'microstrip.xml'; % Archivo de Geometría

% Guarda la geometría, materiales, malla y configuración FDTD
WriteOpenEMS([Sim_Path '/' Sim_CSX], FDTD, CSX);
CSXGeomPlot([Sim_Path '/' Sim_CSX]);
%Ejecuta la simulación
RunOpenEMS(Sim_Path, Sim_CSX);

% Se definen puntos de frecuencia
freq_fdtd = linspace(f_min, f_max, 201);
% Calcular los parámetros de puerto
port = calcPort(port, Sim_Path, freq_fdtd);
% Parámetros S
S11_fdtd = port{1}.uf.ref ./ port{1}.uf.inc;
S21_fdtd = port{2}.uf.ref ./ port{1}.uf.inc;
% Impedancia de entrada
Z_in_fdtd = port{1}.uf.tot ./ port{1}.if.tot;


% COMPARACIÓN Y EVALUACIÓN
fprintf('Análisis Integridad de Potencia (PI)\n');

% Interpolación para comparación
freq_common = linspace(f_min, f_max, 201);
S11_analytical_interp = interp1(freq_analytical, S11_analytical, freq_common);
S21_analytical_interp = interp1(freq_analytical, S21_analytical, freq_common);
Z_in_analytical_interp = interp1(freq_analytical, Z_in_analytical, freq_common);

% Métricas en frecuencia de trabajo
[~, idx_work] = min(abs(freq_common - f_work));

% Parámetros S en frecuencia de trabajo
S11_work_dB = 20*log10(abs(S11_analytical_interp(idx_work)));
S21_work_dB = 20*log10(abs(S21_analytical_interp(idx_work)));

% VSWR
VSWR_analytical_interp = (1 + abs(S11_analytical_interp)) ./ (1 - abs(S11_analytical_interp));
VSWR_work = VSWR_analytical_interp(idx_work);
max_vswr = max(VSWR_analytical_interp);


% Mostrar resultados
fprintf('\n Resultados:\n');
fprintf('Impedancia característica: %.1f Ω\n', Z0_analytical);
fprintf('Impedancia @ 50 kHz: %.1f Ω\n', real(Z_in_analytical_interp(idx_work)));
fprintf('S11 @ 50 kHz: %.1f dB\n', S11_work_dB);
fprintf('S21 @ 50 kHz: %.2f dB\n', S21_work_dB);
fprintf('VSWR @ 50 kHz: %.2f\n', VSWR_work);
fprintf('VSWR máximo: %.2f (%s)\n', max_vswr, vswr_rating);
fprintf('Correlación métodos: %s (error: %.1f Ω)\n', correlation_status, impedance_error);

%% GRÁFICAS
figure('Position', [100, 100, 1400, 800], 'Name', 'Análisis Power Integrity');

% Parámetros S
subplot(2,3,1);
semilogx(freq_common/1000, 20*log10(abs(S11_analytical_interp)), 'b-', 'LineWidth', 3);
hold on;
semilogx(freq_common/1000, 20*log10(abs(S11_fdtd)), 'r--', 'LineWidth', 2);
grid on;
xlabel('Frecuencia (kHz)');
ylabel('S_{11} (dB)');
title('Parámetro S_{11}');
legend('Analítico', 'FDTD', 'Location', 'best');
ylim([-80, 0]);

subplot(2,3,2);
semilogx(freq_common/1000, 20*log10(abs(S21_analytical_interp)), 'b-', 'LineWidth', 3);
hold on;
semilogx(freq_common/1000, 20*log10(abs(S21_fdtd)), 'r--', 'LineWidth', 2);
grid on;
xlabel('Frecuencia (kHz)');
ylabel('S_{21} (dB)');
title('Parámetro S_{21}');
legend('Analítico', 'FDTD', 'Location', 'best');

% Impedancia de entrada
subplot(2,3,3);
semilogx(freq_common/1000, real(Z_in_analytical_interp), 'b-', 'LineWidth', 3);
hold on;
semilogx(freq_common/1000, real(Z_in_fdtd), 'r--', 'LineWidth', 2);
semilogx([f_work f_work]/1000, ylim, 'k:', 'LineWidth', 1);
grid on;
xlabel('Frecuencia (kHz)');
ylabel('Impedancia (Ω)');
title('Impedancia de Entrada');
legend('Analítico', 'FDTD', '50 kHz', 'Location', 'best');

% VSWR
subplot(2,3,4);
semilogx(freq_common/1000, VSWR_analytical_interp, 'b-', 'LineWidth', 3);
hold on;
VSWR_fdtd = (1 + abs(S11_fdtd)) ./ (1 - abs(S11_fdtd));
semilogx(freq_common/1000, VSWR_fdtd, 'r--', 'LineWidth', 2);
semilogx([f_work f_work]/1000, ylim, 'k:', 'LineWidth', 1);
yline(2.0, 'g--', 'VSWR=2.0', 'LineWidth', 1);
grid on;
xlabel('Frecuencia (kHz)');
ylabel('VSWR');
title('Voltage Standing Wave Ratio');
legend('Analítico', 'FDTD', '50 kHz', 'Location', 'best');
ylim([1, 3]);

% Pérdidas totales
subplot(2,3,5);
semilogx(freq_analytical/1000, alpha_total*L_trace*8.686, 'b-', 'LineWidth', 3);
grid on;
xlabel('Frecuencia (kHz)');
ylabel('Pérdidas (dB)');
title('Pérdidas de Inserción');
